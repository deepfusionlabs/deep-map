using Core.Domain;

namespace App;

public class Dispatch
{
    public void Run(IHandler<Customer> h, Customer c, IRepository<Order> orders, IGreeter g, VirtualRepo vr)
    {
        h.Handle(c);
        orders.Get(1);
        g.Greet();
        g.Name();
        vr.Get(2);
    }

    public void Generic<TH, TR>(TH h, TR r, Customer c) where TH : IHandler<Customer> where TR : IRepository<Customer>
    {
        h.Handle(c);
        r.Get(1);
    }
